% All x_T appearances below are retained from the printed source.
% The exceptional empty OR value is stated separately before Eq(2) in the PDF.
I_{\mathrm{and}}(S\mid\boldsymbol{x})
 :=\sum_{T\subseteq S}(-1)^{|S|-|T|}v(\boldsymbol{x}_T),
\qquad I_{\mathrm{and}}(\emptyset\mid\boldsymbol{x})=v(\boldsymbol{x}_\emptyset).

I_{\mathrm{or}}(S\mid\boldsymbol{x})
 :=-\sum_{T\subseteq S}(-1)^{|S|-|T|}v(\boldsymbol{x}_{N\setminus T}),
\qquad I_{\mathrm{or}}(\emptyset\mid\boldsymbol{x})=v(\boldsymbol{x}_\emptyset).

% Theorem 2: main p3 Eq(3), Appendix C p12 Eq(7).
\begin{aligned}
v(\boldsymbol{x}_T)
 &=v_{\mathrm{and}}(\boldsymbol{x}_T)+v_{\mathrm{or}}(\boldsymbol{x}_T)\\
 &=\sum_{S\subseteq T}I_{\mathrm{and}}(S\mid\boldsymbol{x}_T)
   +\sum_{S\in\{S:S\cap T\ne\emptyset\}\cup\{\emptyset\}}
     I_{\mathrm{or}}(S\mid\boldsymbol{x}_T).
\end{aligned}

% Appendix C(1) first states the fixed-x AND sub-conclusion, p12.
\forall T\subseteq N,\qquad
v_{\mathrm{and}}(\boldsymbol{x}_T)=\sum_{S\subseteq T}I_{\mathrm{and}}(S\mid\boldsymbol{x}).

% Appendix C defines interactions for the AND part, pp12--13.
I_{\mathrm{and}}(S\mid\boldsymbol{x})
 :=\sum_{L\subseteq S}(-1)^{|S|-|L|}v_{\mathrm{and}}(\boldsymbol{x}_L).

% The LHS in the displayed Eq(8) prints x_T; retain that original notation.
\begin{aligned}
\sum_{S\subseteq T}I_{\mathrm{and}}(S\mid\boldsymbol{x}_T)
 &=\sum_{S\subseteq T}\sum_{L\subseteq S}(-1)^{|S|-|L|}v_{\mathrm{and}}(\boldsymbol{x}_L)\\
 &=\sum_{L\subseteq T}\sum_{S:L\subseteq S\subseteq T}(-1)^{|S|-|L|}v_{\mathrm{and}}(\boldsymbol{x}_L)\\
 &=v_{\mathrm{and}}(\boldsymbol{x}_T)
   +\sum_{L\subseteq T,L\ne T}v_{\mathrm{and}}(\boldsymbol{x}_L)
     \sum_{m=0}^{|T|-|L|}\binom{|T|-|L|}{m}(-1)^m\\
 &=v_{\mathrm{and}}(\boldsymbol{x}_T).
\end{aligned}

% Appendix C(2), pp13--14: original intermediate claim in case (3).
% This line is a source transcription, not a project endorsement.
L\cap T\ne\emptyset,\quad L\ne N:\qquad
\sum_{S':S'\subseteq N\setminus T\setminus L}
 \underbrace{\sum_{|S''|=0}^{|T|-|T\cap L|}
 \binom{|T|-|T\cap L|}{|S''|}(-1)^{|S'|+|S''|}}_{=0}.

% Appendix C(2) displayed Eq(9), p14.
\sum_{S:S\cap T\ne\emptyset}I_{\mathrm{or}}(S\mid\boldsymbol{x}_T)
 =v_{\mathrm{or}}(\boldsymbol{x}_T)-v_{\mathrm{or}}(\boldsymbol{x}_\emptyset).
